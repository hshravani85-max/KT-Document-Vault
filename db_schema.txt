--- TABLES ---\ncustomers\nCREATE TABLE customers (
            id TEXT PRIMARY KEY,
            name TEXT,
            customer_type TEXT,
            status TEXT
        , mobile TEXT, email TEXT)\n--------------------\ntravellers\nCREATE TABLE travellers (
            id TEXT PRIMARY KEY,
            customer_id TEXT,
            name TEXT,
            type TEXT,
            role TEXT, dob TEXT, relationship TEXT, passport_number TEXT, nationality TEXT,
            FOREIGN KEY(customer_id) REFERENCES customers(id)
        )\n--------------------\ntrips\nCREATE TABLE trips (
            id TEXT PRIMARY KEY,
            customer_id TEXT,
            name TEXT,
            destination TEXT,
            start_date TEXT,
            end_date TEXT,
            trip_type TEXT,
            service TEXT,
            department TEXT,
            status TEXT,
            FOREIGN KEY(customer_id) REFERENCES customers(id)
        )\n--------------------\ntrip_travellers\nCREATE TABLE trip_travellers (
            trip_id TEXT,
            traveller_id TEXT,
            PRIMARY KEY(trip_id, traveller_id),
            FOREIGN KEY(trip_id) REFERENCES trips(id),
            FOREIGN KEY(traveller_id) REFERENCES travellers(id)
        )\n--------------------\nrequirements\nCREATE TABLE requirements (
            id TEXT PRIMARY KEY,
            trip_id TEXT,
            traveller_id TEXT,
            name TEXT,
            type TEXT,
            UNIQUE(trip_id, traveller_id, name)
        )\n--------------------\ndocuments\nCREATE TABLE documents (
            id TEXT PRIMARY KEY,
            requirement_id TEXT,
            status TEXT,
            extracted_data TEXT,
            file_path TEXT,
            uploaded_at TEXT,
            FOREIGN KEY(requirement_id) REFERENCES requirements(id)
        )\n--------------------\nuser_profile\nCREATE TABLE user_profile (
            customer_id TEXT PRIMARY KEY,
            full_name TEXT,
            dob TEXT,
            nationality TEXT,
            passport_number TEXT,
            passport_expiry TEXT
        )\n--------------------\nactivity\nCREATE TABLE activity (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            trip_id TEXT,
            description TEXT,
            timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
        , customer_id TEXT)\n--------------------\nsqlite_sequence\nCREATE TABLE sqlite_sequence(name,seq)\n--------------------\ndocument_history\nCREATE TABLE document_history (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                document_id TEXT,
                event_type TEXT,
                previous_status TEXT,
                new_status TEXT,
                actor TEXT,
                reason TEXT,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY(document_id) REFERENCES documents(id)
            )\n--------------------\nusers\nCREATE TABLE users (
            id TEXT PRIMARY KEY,
            username TEXT UNIQUE,
            password_hash BLOB,
            salt BLOB,
            full_name TEXT,
            role TEXT
        )\n--------------------\nsessions\nCREATE TABLE sessions (
            token TEXT PRIMARY KEY,
            user_id TEXT,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            expires_at DATETIME,
            FOREIGN KEY(user_id) REFERENCES users(id)
        )\n--------------------\n
